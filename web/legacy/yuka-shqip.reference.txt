// ─── Helpers ─────────────────────────────────────────────────────────────────
function getScoreColor(score) {
  if (score >= 75) return { bg: "#27ae60", light: "#d5f5e3", label: ALB.excellent, emoji: "🌿" };
  if (score >= 50) return { bg: "#f1c40f", light: "#fef9e7", label: ALB.good, emoji: "😊" };
  if (score >= 25) return { bg: "#e67e22", light: "#fdebd0", label: ALB.mediocre, emoji: "😐" };
  return { bg: "#e74c3c", light: "#fadbd8", label: ALB.poor, emoji: "⚠️" };
}

function computeScore(product) {
  const ns = product.nutriscore_grade;
  if (ns === "a") return 92;
  if (ns === "b") return 72;
  if (ns === "c") return 50;
  if (ns === "d") return 28;
  if (ns === "e") return 10;
  // fallback from nutriments
  const n = product.nutriments || {};
  let score = 60;
  if (n["sugars_100g"] > 15) score -= 15;
  if (n["saturated-fat_100g"] > 5) score -= 15;
  if (n["salt_100g"] > 1.5) score -= 10;
  if (n["fiber_100g"] > 3) score += 10;
  if (n["proteins_100g"] > 8) score += 8;
  return Math.max(5, Math.min(95, score));
}

function NutriBadge({ grade }) {
  const colors = { a: "#27ae60", b: "#8bc34a", c: "#ffc107", d: "#ff9800", e: "#f44336" };
  if (!grade || grade === "not-applicable") return null;
  return (
    <span style={{
      background: colors[grade] || "#999",
      color: "#fff",
      fontFamily: "'Space Grotesk', sans-serif",
      fontWeight: 800,
      fontSize: 13,
      padding: "2px 10px",
      borderRadius: 20,
      letterSpacing: 1,
      textTransform: "uppercase",
    }}>
      NutriScore {grade?.toUpperCase()}
    </span>
  );
}

function ScoreRing({ score }) {
  const color = getScoreColor(score);
  const r = 52;
  const circ = 2 * Math.PI * r;
  const dash = (score / 100) * circ;
  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 6 }}>
      <div style={{ position: "relative", width: 130, height: 130 }}>
        <svg width="130" height="130" style={{ transform: "rotate(-90deg)" }}>
          <circle cx="65" cy="65" r={r} fill="none" stroke="#eee" strokeWidth="10" />
          <circle
            cx="65" cy="65" r={r} fill="none"
            stroke={color.bg} strokeWidth="10"
            strokeDasharray={`${dash} ${circ}`}
            strokeLinecap="round"
            style={{ transition: "stroke-dasharray 1.2s cubic-bezier(.4,0,.2,1)" }}
          />
        </svg>
        <div style={{
          position: "absolute", inset: 0, display: "flex",
          flexDirection: "column", alignItems: "center", justifyContent: "center",
        }}>
          <span style={{ fontSize: 34, fontWeight: 900, color: color.bg, fontFamily: "'Space Grotesk', sans-serif", lineHeight: 1 }}>
            {score}
          </span>
          <span style={{ fontSize: 11, color: "#888", fontFamily: "sans-serif" }}>/100</span>
        </div>
      </div>
      <span style={{
        background: color.bg, color: "#fff", borderRadius: 20,
        padding: "4px 16px", fontSize: 13, fontWeight: 700,
        fontFamily: "'Space Grotesk', sans-serif", letterSpacing: 0.5,
      }}>
        {color.emoji} {color.label}
      </span>
    </div>
  );
}

function NutrientRow({ label, value, unit, warn }) {
  return (
    <div style={{
      display: "flex", justifyContent: "space-between", alignItems: "center",
      padding: "8px 0", borderBottom: "1px solid #f0f0f0",
    }}>
      <span style={{ fontSize: 14, color: "#555", fontFamily: "sans-serif" }}>{label}</span>
      <span style={{
        fontSize: 14, fontWeight: 700,
        color: warn ? "#e74c3c" : "#222",
        fontFamily: "'Space Grotesk', sans-serif",
      }}>
        {value !== undefined && value !== null ? `${parseFloat(value).toFixed(1)} ${unit}` : "—"}
      </span>
    </div>
  );
}

function AdditiveBadge({ additive }) {
  const risk = additive.toLowerCase().includes("e1") || additive.toLowerCase().includes("e2") ? "orange" : "#888";
  return (
    <span style={{
      display: "inline-block", background: "#f5f5f5", border: `1px solid #ddd`,
      borderRadius: 8, padding: "3px 10px", fontSize: 12,
      fontFamily: "monospace", color: "#555", margin: "3px",
    }}>
      {additive}
    </span>
  );
}
